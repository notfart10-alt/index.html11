<script>
// MATRIX RAIN BACKGROUND
const canvas = document.getElementById("matrix");
const ctx = canvas.getContext("2d");

const chars =
    "アイウエオカキクケコ0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ$#@%&";

let fontSize = 16;
let drops = [];

function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;

    const columns = Math.ceil(canvas.width / fontSize);
    drops = Array.from(
        { length: columns },
        () => Math.random() * -canvas.height / fontSize
    );
}

function drawMatrix() {
    ctx.fillStyle = "rgba(0, 0, 0, 0.08)";
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.font = fontSize + "px monospace";

    for (let i = 0; i < drops.length; i++) {
        const char = chars[
            Math.floor(Math.random() * chars.length)
        ];

        ctx.fillStyle = Math.random() > 0.97
            ? "#ffffff"
            : "#00ff66";

        ctx.fillText(
            char,
            i * fontSize,
            drops[i] * fontSize
        );

        if (
            drops[i] * fontSize > canvas.height &&
            Math.random() > 0.975
        ) {
            drops[i] = 0;
        }

        drops[i] += 0.45;
    }

    requestAnimationFrame(drawMatrix);
}

window.addEventListener("resize", resizeCanvas);
resizeCanvas();
drawMatrix();


// FAKE HACKING SEQUENCE
const form = document.getElementById("hackForm");
const usernameInput = document.getElementById("username");
const button = document.getElementById("hackButton");
const terminal = document.getElementById("terminal");

const sleep = ms =>
    new Promise(resolve => setTimeout(resolve, ms));

function addLine(text, className = "") {
    const line = document.createElement("div");
    line.className = "line " + className;
    line.textContent = "> " + text;
    terminal.appendChild(line);
    terminal.scrollTop = terminal.scrollHeight;
}

form.addEventListener("submit", async event => {
    event.preventDefault();

    const username = usernameInput.value.trim();
    if (!username) return;

    button.disabled = true;
    usernameInput.disabled = true;
    button.textContent = "Hacking...";

    terminal.replaceChildren();

    const steps = [
        "Connecting to imaginary server...",
        "Scanning virtual network nodes...",
        "Bypassing fake firewall...",
        "Decrypting simulated data packets...",
        "Running cybersecurity cinematic sequence...",
        "Generating fictional access credentials...",
        "Finalizing simulation..."
    ];

    for (const step of steps) {
        addLine(step);
        await sleep(650 + Math.random() * 500);

        // Decorative random hexadecimal output
        if (Math.random() > 0.35) {
            const fakeCode = Array.from(
                { length: 4 },

                    .toString(16)
                    .padStart(4, "0")
            ).join(" ");

            addLine("0x" + fakeCode.toUpperCase());
            await sleep(250);
        }
    }

    addLine("Target username: " + username);

    addLine("Hack succesful, Sommar83", "success");
